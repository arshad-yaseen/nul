fn shorted(a: bool, value: i64) bool {
    let both = a and value > 0
    return both or (a and value < 10) or !a
}

fn inspected(seed: i64) !i64 {
    let value = try refuse(seed)
    let supplied = refuse(seed) catch 0
    return refuse(seed) catch |err| {
        if err == error.TooMuch {
            return 100
        }
        return -1
    }
}

fn guarded(maybe: ?i64) i64 {
    let fallback = maybe orelse 0
    let value = maybe orelse {
        return -1
    }
    return value
}
